% --- CHAPTER AND SECTION TITLE STYLING (Using titlesec) ---
\titleformat{\chapter}[display]
  {\normalfont\Huge\bfseries\sffamily\color{DarkGreen}} % Format: sans-serif, dark green
  {\chaptertitlename\ \thechapter} % "Chapter X"
  {20pt} % Separation between "Chapter X" and title
  {\Huge\centering} % Title text format (centered)
  [\vspace{10pt}\titlerule\vspace{10pt}] % Rule below title

\titleformat{\section}
  {\normalfont\Large\bfseries\sffamily\color{WarmBrown}} % Format: sans-serif, warm brown
  {\thesection}
  {1em}
  {}

\titleformat{\subsection}
  {\normalfont\large\bfseries\sffamily\color{DarkGrayText}} % Format: sans-serif
  {\thesubsection}
  {1em}
  {}

\titlespacing*{\chapter}{0pt}{0pt}{40pt} % Spacing around chapter title (left, before, after)
\titlespacing*{\section}{0pt}{3.5ex plus 1ex minus .2ex}{2.3ex plus .2ex}
\titlespacing*{\subsection}{0pt}{3.25ex plus 1ex minus .2ex}{1.5ex plus .2ex}

% --- HEADERS AND FOOTERS (Using fancyhdr) ---
\pagestyle{fancy}
\fancyhf{} % Clear all header and footer fields
\fancyhead[LE,RO]{\sffamily\small\thepage} % Page number on Left (Even) and Right (Odd) outer edge
\fancyhead[RE]{\sffamily\small\leftmark} % Chapter title on Right (Even) inner edge
\fancyhead[LO]{\sffamily\small\rightmark} % Section title on Left (Odd) inner edge
\renewcommand{\headrulewidth}{0.4pt} % Rule below header
\renewcommand{\footrulewidth}{0pt}  % No rule above footer

% Adjust leftmark and rightmark for book class (chapter/section)
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\renewcommand{\sectionmark}[1]{\markright{\thesection\ #1}}

% --- LIST CUSTOMIZATION (Using enumitem) ---
% Pandoc often uses \tightlist for compact lists. Define it to avoid errors if not defined by Pandoc's chain.
% This effectively lets enumitem control list spacing.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

\setlist[itemize,1]{label=\textcolor{DarkGreen}{\textbullet}} % Green bullets
\setlist[itemize,2]{label=\textcolor{WarmBrown}{\textendash}}
\setlist[enumerate,1]{label=\arabic*.}
\setlist[enumerate,2]{label=\alph*)}
