% --- CHAPTER & SECTION TITLE STYLING (titlesec) ---
% Chapter styling
\titleformat{\chapter}[display]
    {\normalfont\huge\bfseries\color{ebookPrimary}} % Format for \chapter command
    {\filcenter\chaptertitlename\ \thechapter} % Label (e.g., "Chapter 1") - filcenter for centering
    {20pt} % Vertical space between label and title
    {\filcenter\Huge} % Format for the chapter title itself - filcenter for centering
    [\vspace{20pt}\titlerule\vspace{20pt}] % Code after title (e.g., a rule)

% Section styling
\titleformat{\section}
    {\normalfont\Large\bfseries\color{ebookSecondary}}

    {}
