% If a cover image is specified using the `cover-image` variable in Pandoc,
% it usually tries to fit it on the first page.
% To ensure it's on its own page before the title:
% The \makecover command will be conditional on $cover-image$ being defined.
% Pandoc's default template behavior for cover-image might be sufficient
% when used with the `book` class. If not, this can be used.
\ifdefined\coverimage
\pretitle{
  \thispagestyle{empty}
  \IfFileExists{$coverimage$}
    {\includegraphics[width=\paperwidth,height=\paperheight,keepaspectratio=false]{$coverimage$}}
    {\Huge No Cover Image Found: $coverimage$} % Debug message
  \cleardoublepage
  \thispagestyle{empty} % Ensure title page itself doesn't get headers/footers from fancyhdr yet
}
\fi
