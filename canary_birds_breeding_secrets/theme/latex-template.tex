% Base document class - 'report' is good for books with chapters
\documentclass[11pt, a4paper]{report}

% == PACKAGES ==
\usepackage{geometry}          % For page margins and layout
\usepackage{fontspec}          % For font selection (XeLaTeX/LuaLaTeX)
\usepackage{titlesec}          % For customizing chapter/section titles
\usepackage{titling}           % For customizing the \maketitle command
\usepackage{fancyhdr}          % For custom headers and footers
\usepackage{xcolor}            % For color definitions
\usepackage{graphicx}          % For including images
\usepackage{listings}          % For code blocks
\usepackage{booktabs}          % For professional quality tables
\usepackage{microtype}         % For improved typography (subtle adjustments)
\usepackage{tocbibind}         % To add ToC, LoF, LoT, Bib to ToC
\usepackage{enumitem}          % For list customization
\usepackage{amsmath}           % For advanced math typesetting (if needed)
\usepackage{amsfonts}          % For math fonts (if needed)
\usepackage{amssymb}           % For math symbols (if needed)
\usepackage{hyperref}          % For hyperlinks and PDF metadata
\usepackage{float}             % For better control over float placement (figures, tables)

% Define \tightlist if not already defined (for Pandoc list compatibility)
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% Define \pandocbounded if not already defined (for Pandoc compatibility with some environments)
\providecommand{\pandocbounded}[1]{#1}

\usepackage{grffile} % Handles more complex filenames and paths for graphics (good practice)
\usepackage{etoolbox} % For robust command patching

% Force \includegraphics to output a placeholder and never error due to image content
% This is for simulation purposes when actual image files are dummies / may not be valid
% Use \nolinkurl from hyperref package to safely typeset filenames with special characters like underscores
\renewcommand{\includegraphics}[2][]{\fbox{IMAGE PLACEHOLDER: \nolinkurl{#2}}}

% == PAGE GEOMETRY ==
\geometry{a4paper, margin=1in} % Standard 1-inch margins

% == FONT SETUP ==
% These are variables that Pandoc will fill from -V options
% \setmainfont{$mainfont$}
% \setsansfont{$sansfont$}
% \setmonofont{$monofont$}
% If not using Pandoc's variables, set them directly:
\setmainfont{Latin Modern Roman}
\setsansfont{Latin Modern Sans}
\setmonofont{Latin Modern Mono}[Scale=0.9] % Slightly smaller mono font for code

% == COLOR DEFINITIONS ==
% Define a subtle accent color (e.g., muted gold or soft green)
% Example: Muted Gold: definecolor{AccentColor}{RGB}{181, 154, 85}
% Example: Soft Green: definecolor{AccentColor}{RGB}{122, 168, 100}
\definecolor{AccentColor}{RGB}{181, 154, 85} % Muted Gold
\definecolor{BlockQuoteBg}{gray}{0.95}      % Light gray for blockquote background
\definecolor{CodeBg}{gray}{0.97}            % Very light gray for code block background

% == TITLE PAGE CUSTOMIZATION ==
% Using 'titling' package to customize \maketitle
\pretitle{\begin{center}\Huge\bfseries\sffamily}
\posttitle{\par\end{center}\vskip 0.5em}
\preauthor{\begin{center}\Large\sffamily}
\postauthor{\par\end{center}\vskip 0.5em}
\predate{\begin{center}\large\sffamily}
\postdate{\par\end{center}\vskip 2em}

% The actual title page will be generated by Pandoc using metadata
% but we ensure the styling here. The content comes from 00-title.md
% via Pandoc's -V title, -V author, -V date variables.

\title{$title$}
\author{$author$}
\date{$date$}

% == CHAPTER AND SECTION TITLE FORMATTING ==
% Using 'titlesec' package
\titleformat{\chapter}[display]
  {\normalfont\Huge\bfseries\sffamily\centering\color{AccentColor}} % Chapter title font, centered, accent color
  {\chaptertitlename\ \thechapter} % "Chapter X"
  {20pt} % Space between "Chapter X" and title
  {\Huge\bfseries\sffamily\color{black}} % Title text font, black
\titlespacing*{\chapter}{0pt}{50pt}{40pt} % Left, before, after spacing

\titleformat{\section}
  {\normalfont\Large\bfseries\sffamily}{\thesection}{1em}{}
\titleformat{\subsection}
  {\normalfont\large\bfseries\sffamily}{\thesubsection}{1em}{}
\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries\sffamily}{\thesubsubsection}{1em}{}

% == HEADERS AND FOOTERS ==
\pagestyle{fancy}
\fancyhf{} % Clear all header and footer fields
\fancyhead[L]{\nouppercase{\leftmark}} % Chapter title on left (even pages if twoside)
\fancyhead[R]{\nouppercase{\rightmark}} % Section title on right (odd pages if twoside)
% For one-sided documents (like many ebooks), these might be better:
% \fancyhead[L]{\sffamily\nouppercase{$title$}} % Book title on left
% \fancyhead[R]{\sffamily\nouppercase{\leftmark}} % Chapter name on right
\fancyfoot[C]{\sffamily\thepage} % Page number in center of footer

\renewcommand{\headrulewidth}{0.4pt} % Subtle rule under header
\renewcommand{\footrulewidth}{0pt}   % No rule above footer

% == TABLE OF CONTENTS ==
\usepackage{tocloft} % For more control over ToC appearance (optional, but good for refinement)
\renewcommand{\cftchapfont}{\sffamily\bfseries}
\renewcommand{\cftsecfont}{\sffamily}
\renewcommand{\cftsubsecfont}{\sffamily}
\renewcommand{\cftchappagefont}{\sffamily\bfseries}
% Add dot leaders to ToC entries
\renewcommand{\cftchapleader}{\sffamily\cftdotfill{\cftdotsep}}

% == LIST FORMATTING ==
\setlist{nosep} % Remove extra vertical spacing in lists
\setlist[itemize,1]{label=\textcolor{AccentColor}{\textbullet}} % Custom bullet for first level
\setlist[itemize,2]{label=\textcolor{AccentColor}{\textendash}}  % Custom dash for second level

% == BLOCKQUOTE STYLING ==
\usepackage{mdframed} % For framed environments, can be used for blockquotes
\newmdenv[
  backgroundcolor=BlockQuoteBg,
  linecolor=AccentColor,
  outerlinewidth=1pt,
  roundcorner=5pt,
  innertopmargin=10pt,
  innerbottommargin=10pt,
  innerrightmargin=10pt,
  innerleftmargin=10pt,
  skipabove=\baselineskip,
  skipbelow=\baselineskip
]{customquote}

% Pandoc uses \blockquote environment. Redefine it.
\renewenvironment{quote}
  {\begin{customquote}\itshape}
  {\end{customquote}}

% == CODE BLOCK STYLING (listings) ==
% Pandoc's --highlight-style option usually handles this well.
% If manual styling is preferred:
\lstdefinestyle{mystyle}{
    backgroundcolor=\color{CodeBg},
    commentstyle=\color{gray}\itshape,
    keywordstyle=\color{blue}\bfseries,
    numberstyle=\tiny\color{gray},
    stringstyle=\color{purple},
    basicstyle=\ttfamily\footnotesize,
    breakatwhitespace=false,
    breaklines=true,
    captionpos=b,
    keepspaces=true,
    numbers=left,
    numbersep=5pt,
    showspaces=false,
    showstringspaces=false,
    showtabs=false,
    tabsize=2,
    frame=tb, % Top and bottom frame lines
    framerule=0.4pt,
    rulecolor=\color{AccentColor}
}
\lstset{style=mystyle}
% Pandoc will use its own listings setup if --listings is used.
% If you use --highlight-style=pygments, this might be overridden or conflict.
% For --highlight-style=... to work best, often less manual lstset is better.
% This template assumes Pandoc handles the specifics if a highlight style is given.

% == TABLES ==
% `booktabs` package is used for \toprule, \midrule, \bottomrule in tables.
% Pandoc will generate tables using these if they are simple enough.

% == FIGURE AND TABLE CAPTIONS ==
\usepackage[font={sf,small},labelfont={bf,sf,color=AccentColor}]{caption}
\captionsetup{skip=10pt} % Space between figure/table and caption

% == HYPERREF SETUP ==
\hypersetup{
    colorlinks=true,
    linkcolor=AccentColor,      % Color of internal links (e.g., toc, cross-references)
    citecolor=AccentColor,      % Color of citation links
    urlcolor=AccentColor,       % Color of URL links
    pdftitle={$title$},
    pdfauthor={$author$},
    pdfsubject={}, % Can add subject if desired
    pdfkeywords={}, % Can add keywords if desired
    bookmarksopen=true,
    bookmarksnumbered=true
}

% == DOCUMENT START ==
\begin{document}

% Generate Title Page (if not using Pandoc's built-in title page mechanism)
% If Pandoc's metadata title page is used (default), this \maketitle is sufficient.
\maketitle

% Generate Table of Contents
\pagenumbering{roman} % Roman numerals for front matter
\tableofcontents
\cleardoublepage % Start main content on a new right-hand page
\pagenumbering{arabic} % Arabic numerals for main content

% == MAIN CONTENT ==
% This is where Pandoc will insert the compiled Markdown content
$body$

% == APPENDIX (if any) ==
% Pandoc can handle appendices if structured correctly in Markdown
% (e.g., using # Appendix A {-})

\end{document}
